const code = pm.response.code;

pm.test("Key not rejected (no 401/403)", () => {
    pm.expect(code, `got HTTP ${code}`).to.not.be.oneOf([401, 403]);
});

pm.test("Status 200", () => pm.response.to.have.status(200));

if (code === 200) {
    const body = pm.response.json();
    const list = body.data || body.models || (Array.isArray(body) ? body : []);
    const ids = list.map(m => m.id || m.name || m);

    pm.test("At least one model available", () => pm.expect(ids.length).to.be.above(0));

    if (ids.length) {
        pm.collectionVariables.set("model", ids[0]);
        console.log("✅ Key valid. Models:", ids.join(", "));
    }
} else if (code === 401 || code === 403) {
    console.log("❌ Key invalid or lacks permission:", pm.response.text());
} else {
    console.log(`⚠️ Unexpected ${code}:`, pm.response.text());
}
